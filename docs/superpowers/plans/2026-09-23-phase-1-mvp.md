# Phase 1 MVP Read-Only & Telegram Bot Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a production-grade, local Python 3.12 trading bot on Windows that authenticates with DNSE OpenAPI via HMAC-SHA256 HTTP Signatures, queries live balance and positions for account `0001880229`, serves an interactive Telegram Bot interface (`aiogram 3.x`) with strict whitelist security, and stores account snapshots in local SQLite.

**Architecture:** Unified Async Event Loop (`asyncio`) running `aiogram 3.x` Telegram polling, `httpx` async DNSE client with Token Bucket rate limiting and in-memory caching, local `aiosqlite` storage, and `APScheduler` background syncing aware of HOSE trading hours.

**Tech Stack:** Python 3.12, `httpx`, `aiogram 3.x`, `pydantic-settings`, `aiosqlite`, `apscheduler`, `pytest`, `pytest-asyncio`.

**Spec:** [`docs/superpowers/specs/2026-09-23-phase-1-mvp-design.md`](file:///d:/TradingBot/docs/superpowers/specs/2026-09-23-phase-1-mvp-design.md)

## Global Constraints

- Python version floor: 3.12+ on Windows.
- DNSE OpenAPI base URL: `https://openapi.dnse.com.vn`.
- Read-only execution gate: Absolutely no order placement, modification, or cancellation logic allowed in this phase.
- Telegram Whitelist: Only user ID `7385637982` is allowed to execute bot commands.
- Security: All credentials (`DNSE_API_KEY`, `DNSE_API_SECRET`, `TELEGRAM_BOT_TOKEN`) must be loaded strictly from `.env` and never logged or committed.
- Test-driven: Every component must have automated tests using `pytest` and `pytest-asyncio`.

## Review Focus

1. **Drift in request timestamp (`Date` header)**: Server requires date within ±1 minute. Must ensure UTC RFC1123 format (`format_datetime(datetime.now(timezone.utc), usegmt=True)`).
2. **URL query parameters in HTTP Signature**: DNSE signing string `(request-target)` MUST omit query parameters (e.g. `/accounts/{id}/positions` not `/accounts/{id}/positions?marketType=STOCK`), tested in `test_signer.py`.
3. **Unauthorized Telegram User Access**: Messages from unknown user IDs must be dropped silently without response or data leakage, tested in `test_whitelist.py`.
4. **Network disconnect / DNSE 5xx errors**: Must trigger Circuit Breaker, log to SQLite `audit_logs`, and serve last known state from in-memory cache instead of crashing.
5. **Windows encoding issues**: Console and log files must handle UTF-8 / Vietnamese characters cleanly without `charmap` UnicodeEncodeErrors.

Write `dnse/signer.py`:
```python
import hmac
import hashlib
import base64
import uuid
from datetime import datetime, timezone
from email.utils import format_datetime

class DNSESigner:
    def __init__(self, api_key: str, api_secret: str):
        self.api_key = api_key
        self.api_secret = api_secret

    def _create_signing_string(self, method: str, path: str, date_str: str, nonce: str) -> str:
        clean_path = path.split('?')[0]
        return f"(request-target): {method.lower()} {clean_path}\ndate: {date_str}\nnonce: {nonce}"

    def build_headers(self, method: str, path: str) -> dict[str, str]:
        now = datetime.now(timezone.utc)
        date_str = format_datetime(now, usegmt=True)
        nonce = uuid.uuid4().hex

        signing_string = self._create_signing_string(method, path, date_str, nonce)
        raw_sig = hmac.new(
            self.api_secret.encode('utf-8'),
            signing_string.encode('utf-8'),
            hashlib.sha256
        ).digest()
        
        b64_sig = base64.b64encode(raw_sig).decode('ascii')
        encoded_sig = b64_sig.replace('+', '%2B').replace('/', '%2F').replace('=', '%3D')

        sig_header = (
            f'Signature keyId="{self.api_key}",'
            f'algorithm="hmac-sha256",'
            f'headers="(request-target) date",'
            f'signature="{encoded_sig}",'
            f'nonce="{nonce}"'
        )

        return {
            "X-API-Key": self.api_key,
            "X-Signature": sig_header,
            "Date": date_str,
            "User-Agent": "DNSE-TradingBot/1.0"
        }
```

import httpx
import logging
from typing import List, Optional
from dnse.signer import DNSESigner
from dnse.models import AccountInfo, AccountBalance, StockBalance, PositionItem

logger = logging.getLogger(__name__)

class DNSEClient:
    def __init__(self, api_key: str, api_secret: str, base_url: str = "https://openapi.dnse.com.vn"):
        self.signer = DNSESigner(api_key, api_secret)
        self.base_url = base_url
        self._client = httpx.AsyncClient(base_url=self.base_url, timeout=10.0)

    async def close(self):
        await self._client.aclose()

    async def get_accounts(self) -> AccountInfo:
        path = "/accounts"
        headers = self.signer.build_headers("GET", path)
        resp = await self._client.get(path, headers=headers)
        resp.raise_for_status()
        return AccountInfo.model_validate(resp.json())

    async def get_balances(self, account_no: str) -> AccountBalance:
        path = f"/accounts/{account_no}/balances"
        headers = self.signer.build_headers("GET", path)
        resp = await self._client.get(path, headers=headers)
        resp.raise_for_status()
        data = resp.json()
        stock_data = data.get("stock", {})
        return AccountBalance(stock=StockBalance.model_validate(stock_data), raw=data)

    async def get_positions(self, account_no: str) -> List[PositionItem]:
        path = f"/accounts/{account_no}/positions"
        query_path = f"{path}?marketType=STOCK"
        headers = self.signer.build_headers("GET", path)
        resp = await self._client.get(query_path, headers=headers)
        resp.raise_for_status()
        data = resp.json()
        raw_items = data.get("positions", [])
        results = []
        for item in raw_items:
            trade_qty = item.get("tradeQuantity", 0) or item.get("openQuantity", 0)
            cost = item.get("costPrice", 0.0)
            market = item.get("marketPrice", cost)
            pnl = (market - cost) * trade_qty
            pnl_pct = ((market - cost) / cost * 100) if cost > 0 else 0.0
            results.append(PositionItem(
                id=item.get("id"),
                symbol=item.get("symbol", ""),
                accountNo=account_no,
                status=item.get("status", "OPEN"),
                side=item.get("side", "NB"),
                tradeQuantity=trade_qty,
                openQuantity=item.get("openQuantity", 0),
                costPrice=cost,
                marketPrice=market,
                unrealizedPnl=pnl,
                pnlPercent=pnl_pct
            ))
        return results
```

Write `storage/repository.py`:

import aiosqlite
from typing import Optional, List, Dict, Any

class Repository:
    def __init__(self, db_path: str = "storage/bot.db"):
        self.db_path = db_path

    async def save_balance_snapshot(self, account_no: str, total_cash: float, available_cash: float, total_debt: float, withdrawable_cash: float):
        async with aiosqlite.connect(self.db_path) as db:
            await db.execute(
                """INSERT INTO account_snapshots (account_no, total_cash, available_cash, total_debt, withdrawable_cash)
                   VALUES (?, ?, ?, ?, ?)""",
                (account_no, total_cash, available_cash, total_debt, withdrawable_cash)
            )
            await db.commit()

    async def get_latest_snapshot(self, account_no: str) -> Optional[Dict[str, Any]]:
        async with aiosqlite.connect(self.db_path) as db:
            db.row_factory = aiosqlite.Row
            async with db.execute(
                "SELECT * FROM account_snapshots WHERE account_no = ? ORDER BY id DESC LIMIT 1",
                (account_no,)
            ) as cursor:
                row = await cursor.fetchone()
                return dict(row) if row else None

    async def log_audit(self, level: str, source: str, message: str):
        async with aiosqlite.connect(self.db_path) as db:
            await db.execute(
                "INSERT INTO audit_logs (level, source, message) VALUES (?, ?, ?)",
                (level, source, message)
            )
            await db.commit()

    async def get_recent_logs(self, limit: int = 10) -> List[Dict[str, Any]]:
        async with aiosqlite.connect(self.db_path) as db:
            db.row_factory = aiosqlite.Row
            async with db.execute(
                "SELECT * FROM audit_logs ORDER BY id DESC LIMIT ?",
                (limit,)
            ) as cursor:
                rows = await cursor.fetchall()
                return [dict(r) for r in rows]
```

from zoneinfo import ZoneInfo

class MarketHours:
    VN_TZ = ZoneInfo("Asia/Ho_Chi_Minh")

    @classmethod
    def get_session(cls, dt: datetime = None) -> str:
        if dt is None:
            dt = datetime.now(cls.VN_TZ)
        elif dt.tzinfo is None:
            dt = dt.replace(tzinfo=cls.VN_TZ)
        else:
            dt = dt.astimezone(cls.VN_TZ)

        # 0 = Monday, 6 = Sunday
        if dt.weekday() >= 5:
            return "CLOSED"

        t = dt.time()
        if t < time(9, 0):
            return "PRE_MARKET"
        elif time(9, 0) <= t < time(9, 15):
            return "ATO"
        elif time(9, 15) <= t < time(11, 30):
            return "CONTINUOUS"
        elif time(11, 30) <= t < time(13, 0):
            return "LUNCH_BREAK"
        elif time(13, 0) <= t < time(14, 30):
            return "CONTINUOUS"
        elif time(14, 30) <= t < time(14, 45):
            return "ATC"
        else:
            return "CLOSED"

    @classmethod
    def is_trading_time(cls, dt: datetime = None) -> bool:
        session = cls.get_session(dt)
        return session in ("ATO", "CONTINUOUS", "ATC")
```

import logging
from aiogram import BaseMiddleware
from aiogram.types import TelegramObject

logger = logging.getLogger(__name__)

class WhitelistMiddleware(BaseMiddleware):
    def __init__(self, admin_id: int):
        self.admin_id = admin_id

    async def __call__(self, handler, event: TelegramObject, data: dict):
        user = getattr(event, "from_user", None)
        if user and user.id == self.admin_id:
            return await handler(event, data)
        logger.warning(f"Unauthorized access attempt blocked from Telegram ID: {user.id if user else 'Unknown'}")
        return None

import logging
import time
from apscheduler.schedulers.asyncio import AsyncIOScheduler
from core.market_hours import MarketHours

logger = logging.getLogger(__name__)

async def sync_data_job(client, state, repo, notifier=None):
    if not client or not state.account:
        return
    sub_id = state.account.accounts[0].id
    try:
        start = time.time()
        bal = await client.get_balances(sub_id)
        pos = await client.get_positions(sub_id)
        state.latency_ms = (time.time() - start) * 1000
        state.set_balance(bal)
        state.set_positions(pos)
        await repo.save_balance_snapshot(
            sub_id, bal.stock.totalCash, bal.stock.availableCash, bal.stock.totalDebt, bal.stock.withdrawableCash
        )
        logger.debug(f"Background sync complete ({state.latency_ms:.1f}ms)")
    except Exception as e:
        logger.error(f"Background sync error: {e}")
        await repo.log_audit("ERROR", "SCHEDULER", f"Sync failed: {e}")

async def daily_summary_job(state, notifier):
    if not notifier or not state.balance:
        return
    bal = state.balance.stock
    acc = state.account
    msg = (
        f"📊 **TỔNG KẾT PHIÊN GIAO DỊCH (15:05)**\n"
        f"• Tài khoản: `{acc.custodyCode if acc else 'N/A'}`\n"
        f"• Tiền mặt: `{bal.totalCash:,.0f} đ`\n"
        f"• Số vị thế nắm giữ qua đêm: {len(state.positions)} mã"
    )
    await notifier.send_alert("Tổng kết cuối phiên", msg, level="INFO")

def setup_scheduler(client, state, repo, notifier=None) -> AsyncIOScheduler:
    scheduler = AsyncIOScheduler()
    # Periodic sync: every 60 seconds
    scheduler.add_job(sync_data_job, "interval", seconds=60, args=[client, state, repo, notifier])
    # Daily market summary at 15:05 Mon-Fri
    scheduler.add_job(daily_summary_job, "cron", day_of_week="mon-fri", hour=15, minute=5, args=[state, notifier])
    return scheduler
```
